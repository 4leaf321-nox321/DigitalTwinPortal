/*******************************************
 * 3) 실행계획/과제 관리용 — 간단 Gantt
 *******************************************/
function daysBetween(a, b) {
  return Math.round((b.getTime() - a.getTime()) / (1000 * 60 * 60 * 24));
}

function ExecutionPlanGantt() {
  // 샘플 태스크
  const tasks = [
    { id: "T1", name: "요건/범위 정리", owner: "사무국", start: "2025-02-01", end: "2025-03-15", deps: [] },
    { id: "T2", name: "데이터 연결 설계", owner: "IT/PLM", start: "2025-03-01", end: "2025-05-15", deps: ["T1"] },
    { id: "T3", name: "센서/수집 파이프라인", owner: "제조기술", start: "2025-04-01", end: "2025-06-30", deps: ["T1"] },
    { id: "T4", name: "PoC 모델 개발", owner: "CAE/AI", start: "2025-05-01", end: "2025-08-31", deps: ["T2","T3"] },
    { id: "T5", name: "PoC 검증·보고", owner: "사무국", start: "2025-09-01", end: "2025-10-15", deps: ["T4"] },
  ];

  const minStart = useMemo(() => new Date("2025-02-01"), []);
  const maxEnd = useMemo(() => new Date("2025-10-31"), []);
  const totalDays = daysBetween(minStart, maxEnd) + 1;

  // 픽셀 스케일 (반응형)
  const usableWidth = 1000; // 캔버스 가로 기준
  const pxPerDay = usableWidth / totalDays;
  const rowH = 36;

  const today = new Date();
  const showToday = today >= minStart && today <= maxEnd;

  return (
    <div className="w-full">
      <h2 className="text-2xl font-bold mb-2">실행계획/과제 관리용 (간단 Gantt)</h2>
      <p className="text-gray-600 mb-4">드래그 편집은 없지만, 기간·의존관계·담당 표시로 운영 회의에 충분</p>

      <div className="overflow-auto bg-white rounded-2xl shadow">
        <div className="min-w-[1100px]">
          {/* 헤더: 날짜 눈금 (월 단위) */}
          <div className="px-4 py-2 border-b bg-gray-50 text-sm text-gray-700 flex items-center">
            <div className="w-56 font-semibold">과제</div>
            <div className="flex-1 relative h-6">
              {/* 월 블럭 */}
              {Array.from({ length: 10 }, (_, i) => {
                const d = new Date(minStart);
                d.setMonth(d.getMonth() + i);
                const monthStart = new Date(d.getFullYear(), d.getMonth(), 1);
                const monthEnd = new Date(d.getFullYear(), d.getMonth() + 1, 0);
                const left = Math.max(0, daysBetween(minStart, monthStart)) * pxPerDay;
                const width = (daysBetween(monthStart, monthEnd) + 1) * pxPerDay;
                return (
                  <div key={i} className="absolute top-0 h-6 border-l border-gray-200" style={{ left, width }}>
                    <div className="text-[11px] px-1 whitespace-nowrap">
                      {d.getFullYear()}-{String(d.getMonth() + 1).padStart(2, "0")}
                    </div>
                  </div>
                );
              })}
              {/* Today 라인 */}
              {showToday && (
                <div className="absolute top-0 bottom-0 w-[2px] bg-sky-500/70" style={{ left: daysBetween(minStart, today) * pxPerDay }} />
              )}
            </div>
          </div>

          {/* 바디: 태스크 행 */}
          {tasks.map((t, idx) => {
            const s = new Date(t.start);
            const e = new Date(t.end);
            const left = Math.max(0, daysBetween(minStart, s)) * pxPerDay;
            const width = Math.max(6, (daysBetween(s, e) + 1) * pxPerDay);
            return (
              <div key={t.id} className="flex items-center border-b last:border-b-0">
                <div className="w-56 px-4 py-2 text-sm">
                  <div className="font-medium">{t.name}</div>
                  <div className="text-gray-500">{t.owner}</div>
                </div>
                <div className="flex-1 relative h-[${rowH}px]">
                  <div className="absolute top-1 h-7 rounded-md shadow" style={{ left, width, background: "#e0f2fe" }} />
                  {/* 의존선 간단 표기 (끝점 점) */}
                  {t.deps?.map((dep) => {
                    const from = tasks.find((x) => x.id === dep);
                    if (!from) return null;
                    const fx = Math.max(0, daysBetween(minStart, new Date(from.end))) * pxPerDay;
                    const tx = left;
                    const y = 26 + idx * rowH;
                    return (
                      <svg key={dep} className="absolute inset-0 pointer-events-none" width={usableWidth} height={(tasks.length + 1) * rowH}>
                        <circle cx={fx} cy={y} r={3} fill="#94a3b8" />
                        <line x1={fx} y1={y} x2={tx} y2={y} stroke="#94a3b8" strokeDasharray="4 2" />
                      </svg>
                    );
                  })}
                </div>
              </div>
            );
          })}
        </div>
      </div>

      <div className="mt-3 text-xs text-gray-500">* 필요 시 dhtmlx-gantt 같은 전문 컴포넌트로 교체하면 드래그 편집·자원배분까지 확장 가능합니다.</div>
    </div>
  );
}
